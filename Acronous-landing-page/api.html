cURL

curl https://api.acronous.com/v1/chat \
  -H "Content-Type: application/json" \
  -H "X-Api-Key: ak_YOUR_KEY" \
  -d '{"message":"Hello!","session_id":"demo"}'

Python

import requests
r = requests.post(
  "https://api.acronous.com/v1/chat",
  headers={"X-Api-Key": "ak_YOUR_KEY"},
  json={"message": "Hello!", "session_id": "demo"},
  timeout=120,
)
print(r.json()["response"])

JavaScript

const r = await fetch("https://api.acronous.com/v1/chat", {
  method: "POST",
  headers: { "Content-Type": "application/json",
             "X-Api-Key": "ak_YOUR_KEY" },
  body: JSON.stringify({ message: "Hello!", session_id: "demo" }),
});
console.log((await r.json()).response);
